cmake_minimum_required(VERSION 3.30)
project(AheadscaleVpnPrototype LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Требуется MSVC и Windows SDK")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(vpn_common STATIC src/ipc_probe.cpp src/app_service.cpp)
target_compile_definitions(vpn_common PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(vpn_common PUBLIC /EHsc /utf-8 /W4 /permissive- /bigobj)
target_link_libraries(vpn_common PUBLIC windowsapp runtimeobject advapi32 userenv)
add_library(AheadscaleVpnPlugin SHARED src/background_task.cpp src/vpn_plugin.cpp src/plugin.def)
target_link_libraries(AheadscaleVpnPlugin PRIVATE vpn_common)
target_link_options(AheadscaleVpnPlugin PRIVATE /APPCONTAINER)
add_executable(AheadscaleVpnController src/profile_controller.cpp)
target_link_libraries(AheadscaleVpnController PRIVATE vpn_common)
add_executable(AheadscaleVpnHost WIN32 src/background_host.cpp src/background_task.cpp src/vpn_plugin.cpp)
target_link_libraries(AheadscaleVpnHost PRIVATE vpn_common)
target_link_options(AheadscaleVpnHost PRIVATE /APPCONTAINER)
